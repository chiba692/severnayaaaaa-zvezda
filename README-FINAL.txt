<script>
const $=s=>document.querySelector(s);const STORE='sz-my-bookings';let token=new URLSearchParams(location.search).get('t')||'',booking=null,chosen='',lastLoadedAt=null;
const statusMap={new:'Ожидает подтверждения',accepted:'Подтверждена',arrived:'Вы в клинике',in_progress:'Идёт приём',completed:'Завершена',rejected:'Отклонена',no_show:'Не пришли',cancelled:'Отменена'};
const serviceMap={exam:'Осмотр',ultrasound:'УЗИ',xray:'Рентген',vaccination:'Вакцинация',tests:'Анализы',inpatient:'Стационар',other:'Другое'};
function esc(s){return String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function dateRu(s){if(!s)return'—';try{return new Intl.DateTimeFormat('ru-RU',{day:'numeric',month:'long',year:'numeric'}).format(new Date(s+'T12:00:00'))}catch{return s}}
async function api(url,opt){const r=await fetch(url,opt),d=await r.json().catch(()=>({}));if(!r.ok)throw Error(d.error||'Ошибка сервера');return d}
function loadSaved(){try{return JSON.parse(localStorage.getItem(STORE)||'[]').filter(x=>x&&x.token).slice(0,5)}catch{return[]}}
function saveList(a){try{localStorage.setItem(STORE,JSON.stringify(a.slice(0,5)))}catch{}}
function remember(tok,b={}){if(!tok)return;const a=loadSaved().filter(x=>x.token!==tok);a.unshift({token:tok,id:b.id||null,pet:b.pet||'Питомец',date:b.booking_date||b.stay_start||'',savedAt:Date.now()});saveList(a)}
function themeSync(){const dark=document.documentElement.dataset.theme==='dark';$('#themeBtn').textContent=dark?'☀':'☾';document.querySelector('meta[name="theme-color"]')?.setAttribute('content',dark?'#101815':'#f5faf7')}
$('#themeBtn').onclick=()=>{const dark=document.documentElement.dataset.theme!=='dark';document.documentElement.dataset.theme=dark?'dark':'';try{localStorage.setItem('sz-theme',dark?'dark':'light')}catch{}themeSync()};themeSync();
function step(s){const order=['new','accepted','arrived','in_progress','completed'],i=order.indexOf(s);return i<0?0:i+1}
function canChange(){return booking&&!['completed','rejected','no_show','cancelled'].includes(booking.status)}
function noticeSeen(){if(!booking?.client_notice_at)return false;try{return localStorage.getItem('sz-notice-'+token)===booking.client_notice_at}catch{return false}}
function acknowledgeNotice(){try{localStorage.setItem('sz-notice-'+token,booking.client_notice_at||'seen')}catch{}render()}
function changeHtml(){if(!booking.client_notice||noticeSeen())return'';return`<div class="changeAlert"><strong>⚠️ Клиника изменила запись</strong><p>${esc(booking.client_notice)}</p><button id="ackChange" class="btn" type="button">Понятно</button></div>`}
function whenText(b){return b.service==='inpatient'?`${dateRu(b.stay_start)} — ${dateRu(b.stay_end)}`:`${dateRu(b.booking_date)} · ${esc(b.booking_time)}`}
function render(){
  const b=booking,s=step(b.status),inpatient=b.service==='inpatient';
  $('#root').innerHTML=`<span class="status ${esc(b.status)}">${esc(statusMap[b.status]||b.status)}</span><h1 class="title">Запись №${esc(b.id)}</h1><p class="sub">${esc(b.owner_name)}, здесь всегда отображается актуальная информация для ${esc(b.pet)}.</p>${changeHtml()}<div class="timeline">${[1,2,3,4,5].map(i=>`<span class="dot ${i<=s?'on':''}"></span>`).join('')}</div><div class="grid"><div class="box"><small>Услуга</small><b>${b.booking_type==='traumatologist'?'Приём травматолога':esc(serviceMap[b.service]||b.service)}</b>${b.booking_type==='traumatologist'?'<small style="margin-top:5px">Стоимость определяется очно после осмотра</small>':''}</div><div class="box"><small>${inpatient?'Период':'Дата и время'}</small><b>${whenText(b)}</b></div><div class="box"><small>Питомец</small><b>${esc(b.pet)}${b.pet_age?' · '+esc(b.pet_age):''}</b></div><div class="box"><small>Тип приёма</small><b>${b.booking_type==='traumatologist'?'Травматолог':'Обычный приём'}</b></div></div>${b.status==='new'?'<div class="notice">Заявка отправлена и ожидает подтверждения клиникой.</div>':''}${b.status==='cancelled'?'<div class="notice error">Эта запись отменена.</div>':''}${b.status==='completed'?'<div class="notice">Приём завершён. Спасибо, что выбрали «Северную звезду».</div>':''}<div class="saved"><span>🔒</span><div><b>Ссылка сохранена на этом устройстве.</b><br>На главной странице появится кнопка «Мои записи», поэтому случайно закрыть эту страницу не страшно.</div></div><div class="actions">${canChange()&&!inpatient?'<button class="btn primary" id="resBtn">Перенести</button>':''}${canChange()?'<button class="btn danger" id="cancelBtn">Отменить запись</button>':''}<button class="btn secondary" id="shareBtn">Поделиться</button><a class="btn secondary" href="tel:+79116480644">Позвонить</a></div><div id="resPanel" class="panel"><b>Выберите новую дату и время</b><input id="newDate" class="input" type="date"><div id="slots" class="slots"><div class="empty">Сначала выберите дату</div></div><button id="saveRes" class="btn primary" style="margin-top:10px" disabled>Отправить перенос</button></div><p class="sync">${lastLoadedAt?'Обновлено '+lastLoadedAt.toLocaleTimeString('ru-RU',{hour:'2-digit',minute:'2-digit'}):''}</p>`;
  if($('#ackChange'))$('#ackChange').onclick=acknowledgeNotice;
  if($('#resBtn'))$('#resBtn').onclick=()=>{$('#resPanel').classList.toggle('show');$('#newDate').min=today()};
  if($('#newDate'))$('#newDate').onchange=loadSlots;
  if($('#saveRes'))$('#saveRes').onclick=reschedule;
  if($('#cancelBtn'))$('#cancelBtn').onclick=cancelBooking;
  if($('#shareBtn'))$('#shareBtn').onclick=shareBooking;
}


function today(){const d=new Date();return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0')}
async function loadSlots(){chosen='';const box=$('#slots');box.innerHTML='<div class="empty">Загружаем…</div>';$('#saveRes').disabled=true;try{const d=await api('/.netlify/functions/get-available-slots?date='+encodeURIComponent($('#newDate').value),{cache:'no-store'});box.innerHTML=d.slots.length?d.slots.map(t=>`<button class="slot" type="button" data-time="${t}">${t}</button>`).join(''):'<div class="empty">Свободного времени нет</div>';document.querySelectorAll('.slot').forEach(b=>b.onclick=()=>{document.querySelectorAll('.slot').forEach(x=>x.classList.remove('on'));b.classList.add('on');chosen=b.dataset.time;$('#saveRes').disabled=false})}catch(e){box.innerHTML='<div class="empty">'+esc(e.message)+'</div>'}}
async function reschedule(){if(!chosen)return;const btn=$('#saveRes');btn.disabled=true;try{await api('/.netlify/functions/public-booking',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({action:'reschedule',token,booking_date:$('#newDate').value,booking_time:chosen})});await load()}catch(e){alert(e.message);btn.disabled=false}}
async function cancelBooking(){if(!confirm('Отменить эту запись? Свободное время снова станет доступно другим клиентам.'))return;try{await api('/.netlify/functions/public-booking',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({action:'cancel',token})});await load()}catch(e){alert(e.message)}}
async function shareBooking(){const url=location.origin+'/booking.html?t='+encodeURIComponent(token);try{if(navigator.share)await navigator.share({title:'Моя запись — Северная звезда',text:`Запись №${booking.id} для ${booking.pet}`,url});else{await navigator.clipboard.writeText(url);alert('Ссылка скопирована')}}catch{}}
async function renderChooser(){
  const a=loadSaved();
  if(!a.length){$('#root').innerHTML='<div class="loading"><b>Сохранённых записей на этом устройстве нет.</b><br><br><a class="btn primary" href="/">Вернуться на сайт</a></div>';return}
  $('#root').innerHTML='<h1 class="title">Мои записи</h1><p class="sub">Проверяем актуальные статусы…</p><div class="chooser"><div class="loading">Загрузка…</div></div>';
  const rows=await Promise.all(a.map(async x=>{try{const d=await api('/.netlify/functions/public-booking?token='+encodeURIComponent(x.token),{cache:'no-store'});return{x,b:d.booking}}catch(e){return{x,error:e.message}}}));
  const ok=rows.filter(r=>r.b);
  if(ok.length)saveList(ok.map(r=>({token:r.x.token,id:r.b.id,pet:r.b.pet,date:r.b.booking_date||r.b.stay_start||'',savedAt:r.x.savedAt||Date.now()})));
  const html=rows.map(r=>{if(r.b){const b=r.b;return `<a class="savedItem" href="/booking.html?t=${encodeURIComponent(r.x.token)}"><div><b>${esc(b.pet||'Питомец')} · №${esc(b.id)}</b><br><small>${esc(statusMap[b.status]||b.status)} · ${dateRu(b.booking_date||b.stay_start)}</small></div><span>→</span></a>`}return `<div class="savedItem"><div><b>${esc(r.x.pet||'Запись')}</b><br><small>Не удалось загрузить: ${esc(r.error||'ошибка')}</small></div><span>!</span></div>`}).join('');
  $('#root').innerHTML=`<h1 class="title">Мои записи</h1><p class="sub">Актуальные записи, сохранённые на этом устройстве.</p><div class="chooser">${html}</div><div class="actions"><button id="retrySaved" class="btn secondary">Обновить</button><a class="btn primary" href="/">Новая запись</a></div>`;
  $('#retrySaved').onclick=renderChooser;
}
async function load(){if(!token){const saved=loadSaved();if(saved.length===1){token=saved[0].token}else{return renderChooser()}}try{const d=await api('/.netlify/functions/public-booking?token='+encodeURIComponent(token),{cache:'no-store'});booking=d.booking;lastLoadedAt=new Date();remember(token,booking);render()}catch(e){$('#root').innerHTML='<div class="loading">'+esc(e.message)+'</div>'}}
load();
document.addEventListener('visibilitychange',()=>{if(!document.hidden&&token)load()});setInterval(()=>{if(!document.hidden&&token&&booking)load()},30000);
