const {reply,env,isAdmin,clean}=require('./_util'); const KEYS=['exam','ultrasound','xray','vaccination','tests','inpatient','other']; exports.handler=async event=>{try{const {url,headers}=env();if(event.httpMethod==='GET'){const r=await fetch(`${url}/rest/v1/clinic_services?select=*&order=sort_order.asc`,{headers});if(!r.ok)throw Error(await r.text());return reply(200,{services:await r.json()})}if(event.httpMethod==='POST'){if(!isAdmin(event))return reply(401,{error:'Требуется авторизация'});const b=JSON.parse(event.body||'{}'),key=clean(b.service_key,30);if(!KEYS.includes(key))return reply(400,{error:'Неизвестная услуга'});const payload={price_label:clean(b.price_label,60),description:clean(b.description,500),prep:clean(b.prep,700),enabled:!!b.enabled,updated_at:new Date().toISOString()};const r=await fetch(`${url}/rest/v1/clinic_services?service_key=eq.${encodeURIComponent(key)}`,{method:'PATCH',headers:{...headers,Prefer:'return=minimal'},body:JSON.stringify(payload)});if(!r.ok)throw Error(await r.text());return reply(200,{ok:true})}return reply(405,{error:'Method not allowed'})}catch(e){console.error(e);return reply(500,{error:'Не удалось загрузить услуги'})}};
